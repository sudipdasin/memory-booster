cmake_minimum_required(VERSION 3.16)
project(MemBoostPro VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MEMBOOST_BUILD_GUI "Build native X11 GUI when X11 headers are available" ON)
include(GNUInstallDirs)
add_library(memboost_core STATIC src/core.cpp)
target_include_directories(memboost_core PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_compile_options(memboost_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(memboost src/main.cpp)
target_link_libraries(memboost PRIVATE memboost_core)
if(MEMBOOST_BUILD_GUI)
  find_package(X11 QUIET)
  if(X11_FOUND)
    target_sources(memboost PRIVATE src/gui.cpp)
    target_link_libraries(memboost PRIVATE X11::X11)
    target_compile_definitions(memboost PRIVATE MEMBOOST_WITH_X11=1)
  else()
    message(WARNING "X11 development headers unavailable; CLI-only binary will be built")
  endif()
endif()
install(TARGETS memboost RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/memboost.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
include(CTest)
if(BUILD_TESTING)
 add_executable(test_core tests/test_core.cpp)
 target_link_libraries(test_core PRIVATE memboost_core)
 target_compile_options(test_core PRIVATE -UNDEBUG)
 add_test(NAME core_unit_tests COMMAND test_core)
 add_test(NAME cli_status COMMAND memboost --status)
 add_test(NAME cli_json COMMAND memboost --json)
 add_test(NAME cli_dry_run COMMAND memboost --apply balanced --dry-run)
endif()
set(CPACK_PACKAGE_NAME "memboost-pro")
set(CPACK_PACKAGE_VENDOR "MemBoost Pro")
set(CPACK_PACKAGE_CONTACT "MemBoost Pro contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Safe native Linux memory monitoring and tuning")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
set(CPACK_GENERATOR "DEB;TGZ")
include(CPack)
