cmake_minimum_required(VERSION 3.16)
project(MemBoostPro VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MEMBOOST_BUILD_GUI "Build native X11 GUI when X11 headers are available" ON)
include(GNUInstallDirs)
add_library(memboost_core STATIC src/core.cpp)
target_include_directories(memboost_core PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_compile_options(memboost_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(memboost src/main.cpp)
target_link_libraries(memboost PRIVATE memboost_core)
target_compile_definitions(memboost PRIVATE MEMBOOST_VERSION="${PROJECT_VERSION}")
set(MEMBOOST_HAS_GUI OFF)
if(MEMBOOST_BUILD_GUI)
  find_package(X11 QUIET)
  if(X11_FOUND)
    target_sources(memboost PRIVATE src/gui.cpp)
    target_link_libraries(memboost PRIVATE X11::X11)
    target_compile_definitions(memboost PRIVATE MEMBOOST_WITH_X11=1)
    set(MEMBOOST_HAS_GUI ON)
  else()
    message(WARNING "X11 development headers unavailable; CLI-only binary will be built")
  endif()
endif()
install(TARGETS memboost RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/memboost-pro)
if(MEMBOOST_HAS_GUI)
 install(FILES packaging/memboost.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(test_core tests/test_core.cpp)
 target_link_libraries(test_core PRIVATE memboost_core)
 target_compile_options(test_core PRIVATE -UNDEBUG)
 add_test(NAME core_unit_tests COMMAND test_core)
 add_test(NAME cli_status COMMAND memboost --status)
 add_test(NAME cli_json COMMAND memboost --json)
 add_test(NAME cli_dry_run COMMAND memboost --apply balanced --dry-run)
 add_test(NAME cli_self_test COMMAND memboost --self-test)
 add_test(NAME cli_diagnose COMMAND memboost --diagnose)
 add_test(NAME cli_signals COMMAND memboost --signals)
 add_test(NAME cli_track_text COMMAND memboost --track --samples 2 --interval-ms 200)
 add_test(NAME cli_track_jsonl COMMAND memboost --track --samples 2 --interval-ms 200 --format jsonl)
 add_test(NAME cli_track_bad_format COMMAND memboost --track --format INVALID)
 set_tests_properties(cli_track_bad_format PROPERTIES WILL_FAIL TRUE)
 add_test(NAME cli_watch_one COMMAND memboost --watch --samples 1 --interval-ms 200)
 add_test(NAME cli_reject_invalid_watch COMMAND memboost --watch --samples 61)
 set_tests_properties(cli_reject_invalid_watch PROPERTIES WILL_FAIL TRUE)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
  add_test(NAME cli_safety_stress COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_cli_safety.py $<TARGET_FILE:memboost>)
 endif()
 add_test(NAME cli_reject_extra_args COMMAND memboost --status --unexpected)
 set_tests_properties(cli_reject_extra_args PROPERTIES WILL_FAIL TRUE)
 add_test(NAME cli_version COMMAND memboost --version)
 set_tests_properties(cli_version PROPERTIES PASS_REGULAR_EXPRESSION "MemBoost Pro ${PROJECT_VERSION}")
 add_test(NAME cli_reject_missing_confirmation COMMAND memboost --apply balanced)
 set_tests_properties(cli_reject_missing_confirmation PROPERTIES WILL_FAIL TRUE)
 add_test(NAME cli_reject_unknown_profile COMMAND memboost --apply nonexistent --yes)
 set_tests_properties(cli_reject_unknown_profile PROPERTIES WILL_FAIL TRUE)
 add_test(NAME cli_reject_invalid_option COMMAND memboost --apply balanced --bad-option)
 set_tests_properties(cli_reject_invalid_option PROPERTIES WILL_FAIL TRUE)
endif()
set(CPACK_PACKAGE_NAME "memboost-pro")
set(CPACK_PACKAGE_VENDOR "MemBoost Pro")
set(CPACK_PACKAGE_CONTACT "MemBoost Pro contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Safe native Linux memory monitoring and tuning")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
set(CPACK_PACKAGE_LICENSE "MIT")
set(CPACK_GENERATOR "DEB;TGZ")
include(CPack)

# Exercise real X11 navigation and native dialog cancellation in a virtual display.
if(BUILD_TESTING AND MEMBOOST_HAS_GUI)
 find_program(MEMBOOST_XVFB_RUN xvfb-run)
 if(MEMBOOST_XVFB_RUN)
  add_executable(test_gui tests/test_gui.cpp)
  target_link_libraries(test_gui PRIVATE X11::X11)
  add_test(NAME gui_interaction COMMAND ${MEMBOOST_XVFB_RUN} -a $<TARGET_FILE:test_gui> $<TARGET_FILE:memboost>)
  set_tests_properties(gui_interaction PROPERTIES TIMEOUT 25)
 endif()
endif()
